The script, kept here as text (it is not part of the builder; it imports requestThroughEnv from the worktree's builder/src/acceptance/browser-fetch.ts):

// One HEAD request through the fetch's own request code (requestThroughEnv), printing who issued the
// certificate the far end showed. Through the proxy's tunnel, TLS runs end to end with the host itself.
import { requestThroughEnv } from "../../wt-31b/builder/src/acceptance/browser-fetch.ts";
const target = new URL(process.argv[2]!);
const ctl = new AbortController();
const t = setTimeout(() => ctl.abort(), 30_000);
const { res, via } = await requestThroughEnv(target, process.env, ctl.signal, "HEAD");
const cert = (res.socket as import("node:tls").TLSSocket).getPeerCertificate?.();
console.log(`${new Date().toISOString()} HEAD ${target.host}: answered ${res.statusCode}, ${via}; content-length ${res.headers["content-length"]}; certificate for ${cert?.subject?.CN}, issued by O=${cert?.issuer?.O}, CN=${cert?.issuer?.CN}`);
res.resume();
clearTimeout(t);
process.exit(0);
