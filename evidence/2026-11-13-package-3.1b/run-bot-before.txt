## 2. A probe of the same refusals, counting what the connection takes after the refusal
# The probe (kept outside the repository, run with npx tsx from builder/), as text:
  | // Probe for the review bot's note on 1ebe1cc: after a refusal at the headers, does the fetch go on reading the body?
  | // A stand-in on 127.0.0.1 answers (a) 200 stating 10^12 bytes, (b) 503 with no stated length; then it sends made-up
  | // bytes as fast as the connection takes them. Counted: the bytes the connection took in the 2 s after the refusal,
  | // and whether it is still open then. Limits given to the fetch: 0.3 s of silence, 0.6 s in all.
  | import http from "node:http";
  | import { mkdtempSync, readdirSync, rmSync } from "node:fs";
  | import { tmpdir } from "node:os";
  | import { join } from "node:path";
  | const { fetchBrowserZip } = await import(join(process.argv[2]!, "src/acceptance/browser-fetch.ts"));
  | const chunk = Buffer.alloc(64 * 1024, 7);
  | let taken = 0;
  | let closed = false;
  | const server = http.createServer((req, res) => {
  |   closed = false;
  |   req.socket.once("close", () => (closed = true));
  |   if (req.url === "/wrong-length") res.writeHead(200, { "content-length": String(10 ** 12) });
  |   else res.writeHead(503);
  |   const more = () => {
  |     while (!res.destroyed) {
  |       taken += chunk.length;
  |       if (!res.write(chunk)) return void res.once("drain", more);
  |     }
  |   };
  |   more();
  | });
  | await new Promise<void>((ok) => server.listen(0, "127.0.0.1", ok));
  | const port = (server.address() as import("node:net").AddressInfo).port;
  | const scratch = mkdtempSync(join(tmpdir(), "drain-probe-"));
  | for (const path of ["/wrong-length", "/refused"]) {
  |   const cache = join(scratch, path.slice(1));
  |   const t0 = Date.now();
  |   const err = await fetchBrowserZip({ cacheDir: cache, testSource: { url: `http://127.0.0.1:${port}${path}`, sha256: "0".repeat(64), bytes: 1000 }, env: {}, silenceLimitMs: 300, timeLimitMs: 600 }).catch((e: Error) => e);
  |   const refusedAt = Date.now();
  |   const atRefusal = taken;
  |   await new Promise((r) => setTimeout(r, 2000));
  |   console.log(`${path}: refused after ${refusedAt - t0} ms: ${err.message}`);
  |   console.log(`  in the 2 s after the refusal the connection took ${((taken - atRefusal) / 1e6).toFixed(1)} MB more; still open at the end: ${!closed}; files left in the cache folder: ${JSON.stringify(readdirSync(cache))}`);
  |   server.closeAllConnections();
  |   await new Promise((r) => setTimeout(r, 100));
  | }
  | rmSync(scratch, { recursive: true, force: true });
  | server.close();
  | process.exit(0);
