#!/usr/bin/env python3
"""Grep proof for package 1.4's renumbering (77-83 -> 91-97) after merging package 1.3's branch.

Run from the repository root of the worktree. Reads git objects only; writes nothing.
"""
import re
import subprocess
import sys

BASE = "4fd2845"          # where package 1.4's branch left package 1.3's
OLD14 = "43351e6"         # package 1.4's head before the merge
B13 = "1171f62"           # package 1.3's head that was merged in
INTEG = "8a0ee2a"         # the integration branch
HEAD = "HEAD"


def git(*args):
    return subprocess.run(["git", *args], capture_output=True, check=True).stdout.decode("utf-8", "replace")


def show(rev, path):
    try:
        return git("show", f"{rev}:{path}")
    except subprocess.CalledProcessError:
        return None


# "gap 91", "gaps 41 and 97", "Gap 93", "known gaps 91 to 97", "gap 91's case", over a line break in a comment too.
REF = re.compile(r"\b[Gg]aps?\b((?:[\s*]*(?:\d+(?:'s)?|to|and|or|,|\(extended\)))+)")
NUM = re.compile(r"\b(\d+)\b")
NOTE = re.compile(r"Numbered (\d+) on package 1\.[34]'s branch")
ROW = re.compile(r"^\| (\d+) \|")


def refs_in(text):
    """Yield (line number, numbers) for each gap reference; a reference that starts at a line's end is
    joined with the next line (a comment's ' * ' is skipped)."""
    lines = text.split("\n")
    for i, line in enumerate(lines):
        joined = line
        if re.search(r"\b[Gg]aps?\s*$", line) and i + 1 < len(lines):
            joined = line + " " + lines[i + 1].lstrip(" */#")
        nums = []
        for m in REF.finditer(joined):
            if m.start() >= len(line):
                continue
            nums += [int(n) for n in NUM.findall(m.group(1))]
        if nums:
            yield i + 1, nums, line


files = git("diff", "--name-only", BASE, OLD14).split()
print(f"Files package 1.4's branch added or changed ({BASE}..{OLD14}): {len(files)}")
print()

old14_rows = range(77, 84)
in_range = lambda n: 67 <= n <= 97
problems = []
ambiguous = []

print("== A. Gap references numbered 67 to 97 in those files, at HEAD")
print("   'same as 1.3' = the line is byte-identical to a line of the same file on package 1.3's branch at", B13)
print("   'this branch' = a line package 1.4's branch wrote or changed")
for f in files:
    text = show(HEAD, f)
    if text is None:
        continue
    b13 = set((show(B13, f) or "").split("\n"))
    for ln, nums, line in refs_in(text):
        hit = [n for n in nums if in_range(n)]
        if not hit:
            continue
        who = "same as 1.3" if line in b13 else "this branch"
        if who == "this branch":
            # 67 to 80 in a line this branch wrote would be an old number (package 1.3's 67 to 76 or this
            # branch's 77 to 80). 81 to 83 are package 1.3's numbers now, so each such line is listed in
            # section A2 to be read.
            bad = [n for n in hit if 67 <= n <= 80]
            if bad:
                problems.append(f"{f}:{ln} this branch's line names {bad}")
            if any(81 <= n <= 90 for n in hit):
                ambiguous.append((f, ln, line))
        if f == "docs/KNOWN-GAPS.md" and ROW.match(line):
            row = int(ROW.match(line).group(1))
            label = f"row {row}"
        else:
            label = ""
        print(f"   {f}:{ln}  gaps {hit}  [{who}] {label}".rstrip())
print()
print("== A2. Lines this branch wrote that name 81 to 90 (package 1.3's numbers now; read each)")
for f, ln, line in ambiguous:
    print(f"   {f}:{ln}: {line}")
print()

print("== B. Former-number notes (the only places the old numbers 77 to 83 of package 1.4 stay)")
kg = show(HEAD, "docs/KNOWN-GAPS.md").split("\n")
for i, line in enumerate(kg):
    m = ROW.match(line)
    if not m:
        continue
    row = int(m.group(1))
    for n in NOTE.finditer(line):
        print(f"   docs/KNOWN-GAPS.md:{i+1}  row {row}: \"Numbered {n.group(1)} on ...\"")
print()

print("== C. Rows of docs/KNOWN-GAPS.md from 60 on, at HEAD")
rows = []
for i, line in enumerate(kg):
    m = ROW.match(line)
    if m and int(m.group(1)) >= 60:
        rows.append(int(m.group(1)))
        print(f"   {i+1}: {line[:110]}")
expected = list(range(60, 70)) + list(range(81, 98))
print(f"   numbers: {rows}")
if rows != expected:
    problems.append(f"rows are {rows}, expected {expected}")
print()

print("== D. Package 1.4's rows: HEAD's 91 to 97 against", OLD14 + "'s 77 to 83, after taking out the number, the note and the one reference changed")
old = {int(ROW.match(l).group(1)): l for l in show(OLD14, "docs/KNOWN-GAPS.md").split("\n") if ROW.match(l)}
new = {int(ROW.match(l).group(1)): l for l in kg if ROW.match(l)}
for o in old14_rows:
    n = o + 14
    a = old[o][len(f"| {o} | "):]
    b = new[n][len(f"| {n} | "):]
    b = b.replace(f" *(Numbered {o} on package 1.4's branch until that branch was brought up to date with package 1.3's branch and the integration branch on 13 Nov 2026.)*", "")
    b = b.replace("gap 91's case", "gap 77's case")
    same = a == b
    print(f"   {o} -> {n}: {'identical otherwise' if same else 'DIFFERS'}")
    if not same:
        problems.append(f"row {n} differs from old row {o} beyond the renumbering")
print()

print("== E. Package 1.3's rows 81 to 90 and the integration branch's 67 to 69: HEAD against", B13)
old13 = {int(ROW.match(l).group(1)): l for l in show(B13, "docs/KNOWN-GAPS.md").split("\n") if ROW.match(l)}
for n in list(range(67, 70)) + list(range(81, 91)):
    same = old13.get(n) == new.get(n)
    print(f"   row {n}: {'byte-identical' if same else 'DIFFERS'}")
    if not same:
        problems.append(f"row {n} differs from package 1.3's branch")
integ_kg = {int(ROW.match(l).group(1)): l for l in show(INTEG, "docs/KNOWN-GAPS.md").split("\n") if ROW.match(l)}
for n in range(67, 70):
    same = integ_kg.get(n) == new.get(n)
    print(f"   row {n} against the integration branch at {INTEG}: {'byte-identical' if same else 'DIFFERS'}")
    if not same:
        problems.append(f"row {n} differs from the integration branch")
print()

print("== F. Every gap reference numbered 91 to 97 anywhere in the repository at HEAD (all tracked text files)")
for f in git("ls-files").split():
    if f.startswith("docs/licences/raw/") or f.endswith((".png", ".jpg", ".webm", ".zip", ".pdf")):
        continue
    text = show(HEAD, f)
    if text is None:
        continue
    for ln, nums, line in refs_in(text):
        hit = [n for n in nums if 91 <= n <= 97]
        if hit:
            print(f"   {f}:{ln}  gaps {hit}")
            if f not in files:
                problems.append(f"{f}:{ln} names {hit} but is not one of package 1.4's files")
print()

print("== G. Any reference to package 1.4's old numbers 77 to 80 at HEAD outside the former-number notes (expected: none)")
count = 0
for f in git("ls-files").split():
    if f.startswith("docs/licences/raw/"):
        continue
    text = show(HEAD, f)
    if text is None:
        continue
    for ln, nums, line in refs_in(text):
        hit = [n for n in nums if 77 <= n <= 80]
        if hit:
            count += 1
            print(f"   {f}:{ln}  gaps {hit}")
            problems.append(f"{f}:{ln} still names {hit}")
print(f"   count: {count}")
print()

print("== H. Test names and run outputs of package 1.4 that name a gap (expected: none)")
count = 0
for f in files:
    if not (f.startswith("evidence/2026-11-13-package-1.4/run-") or f.startswith("builder/test/")):
        continue
    text = show(HEAD, f) or ""
    for ln, nums, line in refs_in(text):
        count += 1
        print(f"   {f}:{ln}  gaps {nums}")
print(f"   count: {count}")
print()

if problems:
    print("PROBLEMS:")
    for p in problems:
        print("  ", p)
    sys.exit(1)
