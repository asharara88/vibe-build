#!/bin/bash
# Reviewer's probe 1: is a container's .Image what it was CREATED from, or is it looked up again by the tag's name when asked?
# Everything made here carries the reviewer's own project id (builder_p_iso06c_rev_real-...), and is removed at the end.
set -u
P=builder_p_iso06c_rev_real
NODE='node:22-alpine@sha256:0e1754758c2df4cfa3e68d59e2e1b7374bb7fd9ea6328eaaacf15cc934af8202'
T=$P-tagmove:latest; B=$P-idcheck:latest; C=$P-idc
cleanup() { docker rm -f "$C" >/dev/null 2>&1; docker image rm -f "$T" "$B" >/dev/null 2>&1; }
cleanup
printf 'FROM %s\nLABEL reviewer.probe="1"\n' "$NODE" | docker build --network none -q -t "$B" - >/dev/null 2>&1 || { echo "build failed"; cleanup; exit 1; }
docker tag "$NODE" "$T"
echo "tag T ($T) at first:        $(docker image inspect "$T" --format '{{.Id}}')"
echo "other image B ($B):          $(docker image inspect "$B" --format '{{.Id}}')"
docker run -d --name "$C" --network none --read-only --cap-drop ALL --entrypoint "" "$T" node -e "setInterval(()=>{},1e6)" >/dev/null
echo "container made from T, .Image: $(docker inspect "$C" --format '{{.Image}}')  .Config.Image: $(docker inspect "$C" --format '{{.Config.Image}}')"
docker tag "$B" "$T"
echo "tag T after it was moved to B: $(docker image inspect "$T" --format '{{.Id}}')"
echo "the SAME container, .Image now: $(docker inspect "$C" --format '{{.Image}}')  (still running: $(docker inspect "$C" --format '{{.State.Running}}'))"
cleanup
echo "left: containers [$(docker ps -a --filter "name=$P-idc" --format '{{.Names}}' | tr '\n' ' ')] images [$(docker images --filter "reference=$P-*" --format '{{.Repository}}:{{.Tag}}' | tr '\n' ' ')]"
