// Part C measurement M2 (package 3.2): scripts/check-export.ts as a command, end to end, against a copy started by its
// README's own command. Scratch script, not in the repository. Made-up data only; no secret is read or printed.
// Every Docker object it makes is removed by exact name at the end; Docker's lists are compared before and after.
import { execFile, spawn } from "node:child_process";
import { randomBytes } from "node:crypto";
import { mkdirSync, mkdtempSync, readFileSync, rmSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { promisify } from "node:util";

const WT = "<scratch>/wt-32/builder";
const { exampleChecklist, standInSources } = await import(`${WT}/src/acceptance/stand-in.ts`);
const { openStore, writeVersion } = await import(`${WT}/src/acceptance/store.ts`);
const { example1Sandboxes } = await import(`${WT}/src/orchestrator/example-1-adapter.ts`);
const { startItCommand, writeExport } = await import(`${WT}/src/orchestrator/export.ts`);
const { runStep } = await import(`${WT}/scripts/example-1-sandbox.ts`);

const run = promisify(execFile);
const say = (s: string) => console.log(`[${new Date().toISOString().slice(11, 19)}] ${s.split(tmpdir()).join("<tmp>")}`);
const plain = { PATH: process.env.PATH ?? "/usr/bin:/bin", HOME: process.env.HOME ?? "/root" };
const copyEnv = { ...plain, APP_PORT: "3192" };
const docker = async (args: string[], o: { cwd?: string; env?: NodeJS.ProcessEnv } = {}) => (await run("docker", args, { env: o.env ?? plain, cwd: o.cwd, maxBuffer: 32 << 20, timeout: 600_000 })).stdout;
const lists = async () => ({
  containers: (await docker(["ps", "-a", "--format", "{{.Names}}"])).split("\n").filter(Boolean).sort(),
  images: (await docker(["images", "-q"])).split("\n").filter(Boolean).sort(),
  volumes: (await docker(["volume", "ls", "-q"])).split("\n").filter(Boolean).sort(),
  networks: (await docker(["network", "ls", "--format", "{{.Name}}"])).split("\n").filter(Boolean).sort(),
});

const PROJECT = `j1-${randomBytes(6).toString("hex")}`;
const work = mkdtempSync(join(tmpdir(), "x32-m2-"));
const before = await lists();
const sbx = example1Sandboxes({ buildProxyPort: 13193 });
let x: Awaited<ReturnType<typeof sbx.make>> | null = null;
let copy = "";
let up: ReturnType<typeof spawn> | null = null;
try {
  say(`project ${PROJECT}; the export made as in L1 (the engine's two steps, a stand-in version, the real export checks)`);
  x = await sbx.make(PROJECT);
  for (const step of ["01-data", "02-submitter-screen"]) if (!(await runStep(x, step)).ok) throw new Error(step);
  mkdirSync(join(work, "data"));
  const store = openStore(join(work, "data"), { refuse: [x.dir] });
  const spec = exampleChecklist("example-1");
  const v = writeVersion(store, PROJECT, {
    writtenBy: "stand-in",
    appSpecSha256: spec.sha256,
    items: standInSources("example-1").map((s: { number: number }) => ({ ...s, checklist: "made-up checklist line (measurement M2)" })),
  });
  const x1 = x;
  const out = await writeExport({ repo: x1.repo, store, projectId: PROJECT, version: v.version, exportsDir: join(work, "data", "journey", PROJECT, "exports"), runManifestSha256: v.manifestSha256, check: (p: never) => sbx.checkExport(x1, p) });
  say(`export written: ${out.bytes} bytes; checks ${out.checks.map((c: { id: string; passed: boolean }) => `${c.id}=${c.passed}`).join(", ")}`);
  // The builder's own sandbox is removed before the copy starts: nothing of it is running during the check.
  await sbx.destroy(x);
  x = null;
  say("the builder's sandbox for this project removed (containers, images, folder)");
  mkdirSync(join(work, "delivered"));
  await run("tar", ["--extract", "--gzip", "--file", out.path, "--directory", join(work, "delivered"), "--no-same-owner"]);
  copy = join(work, "delivered", PROJECT);
  const command = startItCommand(readFileSync(join(copy, "README.md"), "utf8"))!;
  say(`the README's command: ${command} (PATH, HOME and APP_PORT=3192 only)`);
  const argv = command.split(" ");
  up = spawn(argv[0]!, argv.slice(1), { cwd: copy, env: copyEnv, stdio: ["ignore", "ignore", "ignore"] });
  for (let i = 0; i < 300; i++) {
    const ps = (await docker(["compose", "ps", "--format", "json"], { cwd: copy, env: copyEnv }).catch(() => "")).trim();
    const app = ps ? ps.split("\n").map((l) => JSON.parse(l)).find((c: { Service?: string }) => c.Service === "app") : undefined;
    if (app?.State === "running" && app.Health === "healthy") break;
    await new Promise((r) => setTimeout(r, 2000));
  }
  say("the copy's app is healthy; now the script, as a person on this machine would run it:");
  const args = [copy, "--project-id", PROJECT, "--version", v.version, "--manifest-sha256", out.manifestSha256, "--checklist-items", String(spec.items.length), "--run-folder", join(work, "check-run")];
  say(`npx tsx scripts/check-export.ts <copy> --project-id ${PROJECT} --version ${v.version} --manifest-sha256 ${out.manifestSha256} --checklist-items ${spec.items.length} --run-folder <run folder>`);
  const t0 = Date.now();
  const r = await run(process.execPath, ["--import", "tsx", "scripts/check-export.ts", ...args], { cwd: WT, env: { ...plain, TMPDIR: tmpdir() }, maxBuffer: 8 << 20, timeout: 900_000 }).then(
    (o) => ({ code: 0, out: o.stdout + o.stderr }),
    (e: { code?: number; stdout?: string; stderr?: string }) => ({ code: e.code ?? null, out: `${e.stdout ?? ""}${e.stderr ?? ""}` }),
  );
  say(`the script ended with exit code ${r.code} after ${Math.round((Date.now() - t0) / 1000)} s; it printed:`);
  for (const l of r.out.split("\n").filter(Boolean)) say(`  | ${l}`);
} finally {
  if (copy) await docker(["compose", "down", "--volumes", "--remove-orphans"], { cwd: copy, env: copyEnv }).catch(() => undefined);
  if (up && up.exitCode === null) up.kill("SIGTERM");
  if (x) await sbx.destroy(x).catch(() => undefined);
  for (const image of [`${PROJECT}-app`, `${PROJECT}-setup`, `builder_${PROJECT}-exportcheck-run`, `builder_${PROJECT}-exportcheck-tools`]) await docker(["image", "rm", "-f", image]).catch(() => undefined);
  rmSync(work, { recursive: true, force: true });
  const after = await lists();
  const diff = (k: keyof typeof before) => ({ added: after[k].filter((n) => !before[k].includes(n)), removed: before[k].filter((n) => !after[k].includes(n)) });
  say(`Docker's lists against the start: ${JSON.stringify({ containers: diff("containers"), images: diff("images"), volumes: diff("volumes"), networks: diff("networks") })}`);
}
