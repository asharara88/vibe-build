== The script of section 2 (refs.py), as run: it lists each "gap(s)", "Numbered" or "becomes" followed by numbers, expands "N to M", and prints the match when any number is from 67 to 90
```
import re, sys, subprocess
root = sys.argv[1]
files = [l.strip() for l in open(sys.argv[2]) if l.strip() and not l.startswith('evidence/2026-11-13-package-1.3/run-')]
# also scan run outputs, separately reported
runs = [l.strip() for l in open(sys.argv[2]) if l.startswith('evidence/2026-11-13-package-1.3/run-')]
# a "gap reference": 'gap'/'gaps'/'Numbered'/'becomes' followed by a list of numbers, joined by , and to / ;
pat = re.compile(r'(?i)\b(gaps?|numbered|becomes)\b((?:\s*(?:\d+|,|and|to|known gaps?)\s*)+)')
for f in files + runs:
    try: text = open(root + '/' + f).read()
    except FileNotFoundError: continue
    # join wrapped comment lines: track line numbers
    for m in pat.finditer(text):
        nums = [int(x) for x in re.findall(r'\d+', m.group(2))]
        span = ' '.join(m.group(0).split())
        # expand "a to b"
        rng = re.findall(r'(\d+)\s+to\s+(\d+)', m.group(2))
        allnums = set(nums)
        for a, b in rng: allnums |= set(range(int(a), int(b) + 1))
        if any(67 <= n <= 90 for n in allnums):
            line = text.count('\n', 0, m.start()) + 1
            print(f"{f}:{line}: {span}")
```
